//script
let completedTasks = 0;
let points = 0;

const quotes = [
  "You did it! Keep going!",
  "Be Consistent!",
  "Great job! 🎉",
  "Keep up the awesome work!",
  "You make it look easy!"
];

function addTask() {
  const taskInput = document.getElementById("taskInput");
  const taskText = taskInput.value.trim();
  if (!taskText) return;

  const taskList = document.getElementById("taskList");

  const li = document.createElement("li");

  const textSpan = document.createElement("span");
  textSpan.textContent = taskText;

  const buttonDiv = document.createElement("div");
  buttonDiv.className = "task-buttons";

  const doneBtn = document.createElement("button");
  doneBtn.textContent = "✅ Done";
  doneBtn.onclick = () => completeTask(doneBtn);

  const editBtn = document.createElement("button");
  editBtn.textContent = "✏️ Edit";
  editBtn.onclick = () => {
    const newText = prompt("Edit your task:", textSpan.textContent);
    if (newText !== null && newText.trim() !== "") {
      textSpan.textContent = newText.trim();
    }
  };

  const deleteBtn = document.createElement("button");
  deleteBtn.textContent = "🗑️ Delete";
  deleteBtn.onclick = () => deleteTask(deleteBtn);

  buttonDiv.appendChild(doneBtn);
  buttonDiv.appendChild(editBtn);
  buttonDiv.appendChild(deleteBtn);

  li.appendChild(textSpan);
  li.appendChild(buttonDiv);
  taskList.appendChild(li);

  taskInput.value = "";
}

function deleteTask(button) {
  const li = button.closest("li");
  li.remove();
}

function completeTask(button) {
  const li = button.closest("li");
  li.style.textDecoration = "line-through";
  li.style.opacity = "0.6";
  button.disabled = true;

  launchEmojiConfetti();

  points += 1;
  document.getElementById("pointCount").textContent = points;

  completedTasks++;
  if (completedTasks % 3 === 0) {
    showQuote();
  }
}

function launchEmojiConfetti() {
  const emojis = ['🎉', '🌟', '😄', '✨', '🎈', '🎊', '👏', '🦄', '💖'];
  for (let i = 0; i < 30; i++) {
    const emoji = document.createElement("div");
    emoji.className = "emoji";
    emoji.textContent = emojis[Math.floor(Math.random() * emojis.length)];
    emoji.style.left = Math.random() * 100 + "vw";
    emoji.style.top = "100vh";
    emoji.style.position = "absolute";
    emoji.style.fontSize = "24px";
    emoji.style.zIndex = "1000";
    emoji.style.animation = "floatUp 2s ease-out forwards";
    document.body.appendChild(emoji);
    setTimeout(() => emoji.remove(), 2000);
  }
}

function showQuote() {
  const quoteBox = document.getElementById("quoteBox");
  const quoteText = document.getElementById("quoteText");
  quoteText.textContent = quotes[Math.floor(Math.random() * quotes.length)];
  quoteBox.classList.add("show");
  setTimeout(() => quoteBox.classList.remove("show"), 4000);
}
